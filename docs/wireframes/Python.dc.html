Here’s a version that sorts once, then lets later rows overwrite earlier ones:

def dedupe_latest(rows):
    latest = {}
    for row in sorted(rows, key=lambda r: r["updated_at"]):
        latest[row["id"]] = row
    return list(latest.values())
